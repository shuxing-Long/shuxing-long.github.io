def compose(*args):
    def new_function(input):      ←---- 开始定义 compose 返回的
函数
        state = input      ←---- 设置当前的 state 等于 input
        for f in reversed(args):      ←---- 因为组合函数的内部

            state = f(state)      ←---- 在每一步，通过执行下一

        return state
    return new_function为了检查上述工作，我们可以实现一些函数，并将它们组合

def prepend(string):
    def new_function(input):
        return string + input
    return new_function
